### Q1. Thermostat Schedule — min clamp-down ops + max sum (your actual question)
**Problem:** Given array t[1..N]. A clamp-down picks index i and value v ≤ t[i], sets t[j] = min(t[j], v) for all j ≥ i. Find min operations to make array non-decreasing, and max possible sum among all ways achieving that minimum.

**Approach:** Scan right to left, tracking the running minimum `mn`. Whenever `t[i] > mn`, that position *must* be clamped (one operation), and the best value to clamp to is exactly `mn` (the smallest we're forced to go — using anything larger would violate non-decreasing with what follows). Count these forced clamps; for the sum, each forced position becomes `mn` at that point, others stay as-is.

```python
def thermostat_schedule(t):
    n = len(t)
    ops = 0
    total = 0
    suffix_min = float('inf')
    result = [0] * n
    for i in range(n - 1, -1, -1):
        if t[i] > suffix_min:
            ops += 1
            result[i] = suffix_min
        else:
            result[i] = t[i]
            suffix_min = t[i]
    total = sum(result)
    return ops, total

# Example
print(thermostat_schedule([5, 2, 3]))   # (1, 6)
print(thermostat_schedule([1, 5, 2, 6, 3]))  # (1, 9)
print(thermostat_schedule([1, 2, 3, 4]))  # (0, 10)
```
**Complexity:** O(N) time, O(N) space.

### Q5. Trapping Rain Water
```python
def trap(height):
    if not height:
        return 0
    n = len(height)
    left, right = 0, n - 1
    left_max, right_max = height[0], height[-1]
    water = 0
    while left < right:
        if left_max <= right_max:
            left += 1
            left_max = max(left_max, height[left])
            water += left_max - height[left]
        else:
            right -= 1
            right_max = max(right_max, height[right])
            water += right_max - height[right]
    return water

print(trap([0,1,0,2,1,0,1,3,2,1,2,1]))  # 6
```
**Complexity:** O(N) time, O(1) space.

### Q9. Sliding Window Maximum (monotonic deque)

```python
from collections import deque

def sliding_window_max(nums, k):
    dq = deque()  # stores indices, decreasing values
    res = []
    for i, x in enumerate(nums):
        while dq and nums[dq[-1]] <= x:
            dq.pop()
        dq.append(i)
        if dq[0] <= i - k:
            dq.popleft()
        if i >= k - 1:
            res.append(nums[dq[0]])
    return res
```

### Q11. Compositions of N with parts in {1..K}, sum of lengths (your actual question)
**Problem:** Count sum of lengths over all ordered sequences of positive integers, each part ≤ K, summing to N.

**Approach:** Let `cnt[s]` = number of compositions summing to s, `len_sum[s]` = sum of lengths of all compositions summing to s.
Recurrence (choosing the *last* part `p` from 1..K):
```
cnt[s]     = sum(cnt[s-p] for p in 1..K), cnt[0] = 1
len_sum[s] = sum(len_sum[s-p] + cnt[s-p] for p in 1..K)
```
Use a sliding window sum for O(N) instead of O(N·K).

```python
MOD = 10**9 + 7

def sum_of_lengths(N, K):
    cnt = [0] * (N + 1)
    len_sum = [0] * (N + 1)
    cnt[0] = 1
    # sliding window sums over the last K values of cnt / len_sum
    window_cnt = 0
    window_len = 0
    for s in range(1, N + 1):
        window_cnt = (window_cnt + cnt[s - 1]) % MOD
        window_len = (window_len + len_sum[s - 1]) % MOD
        if s - 1 - K >= 0:
            window_cnt = (window_cnt - cnt[s - 1 - K]) % MOD
            window_len = (window_len - len_sum[s - 1 - K]) % MOD
        cnt[s] = window_cnt % MOD
        len_sum[s] = (window_len + window_cnt) % MOD
    return len_sum[N] % MOD

print(sum_of_lengths(4, 2))  # 15
print(sum_of_lengths(5, 3))  # 43
print(sum_of_lengths(6, 6))  # 112
```
**Complexity:** O(N) time, O(N) space.

### Q15. Coin Change II with minimum-gap constraint D between distinct denominations used
```python
def coin_change_min_gap(amount, coins, D):
    MOD = 10**9 + 7
    coins = sorted(set(coins))
    n = len(coins)
    dp = [[0] * (amount + 1) for _ in range(n)]   # dp[i][a]: coins[i] is largest used
    pref = [0] * (amount + 1)
    ptr = 0
    total = 0
    for i in range(n):
        coin = coins[i]
        while ptr < i and coin - coins[ptr] >= D:
            for a in range(amount + 1):
                pref[a] = (pref[a] + dp[ptr][a]) % MOD
            ptr += 1
        for a in range(coin, amount + 1):
            ways = pref[a - coin]
            if (a - coin) % coin == 0:
                ways = (ways + 1) % MOD
            ways = (ways + dp[i][a - coin]) % MOD
            dp[i][a] = ways
        total = (total + dp[i][amount]) % MOD
    return total if amount else 1
```
**Complexity:** O(len(coins) · amount).

### Q19. Number of ways to partition N into exactly K positive parts (unordered)
```python
def partitions_k_parts(N, K):
    dp = [[0] * (N + 1) for _ in range(K + 1)]
    dp[0][0] = 1
    for k in range(1, K + 1):
        for n in range(1, N + 1):
            # each part >= 1: subtract 1 from every part (bijection trick)
            dp[k][n] = dp[k][n - k] if n >= k else 0
            if n == k:
                dp[k][n] += 1
    # simpler direct recurrence using partitions into parts <= given size is more standard;
    # for exact-parts count use recursive definition p(n,k) = p(n-1,k-1) + p(n-k,k)
    def p(n, k, memo={}):
        if n == 0 and k == 0:
            return 1
        if n <= 0 or k <= 0:
            return 0
        if (n, k) in memo:
            return memo[(n, k)]
        memo[(n, k)] = p(n - 1, k - 1, memo) + p(n - k, k, memo)
        return memo[(n, k)]
    return p(N, K)
```
**Complexity:** O(N·K).

### Q21. Edit Distance with a one-time "wildcard block replace" (your actual question)
```python
def min_cost_transform(A, B, K, W):
    n, m = len(A), len(B)
    # standard edit distance (insert/delete only, cost 1 each) prefix[i][j]
    # = cost to convert A[:i] to B[:j] using only insert/delete (i.e. i+j - 2*LCS... 
    # but since only insert/delete allowed, cost = i + j - 2*lcs(A[:i], B[:j]))
    lcs = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if A[i - 1] == B[j - 1]:
                lcs[i][j] = lcs[i - 1][j - 1] + 1
            else:
                lcs[i][j] = max(lcs[i - 1][j], lcs[i][j - 1])

    def indel_cost(i, j):
        return i + j - 2 * lcs[i][j]

    # baseline: no wildcard used
    best = indel_cost(n, m)

    # try using wildcard: consume A[i:i+K] and B[j:j+K] as a block replace
    for i in range(0, n - K + 1):
        for j in range(0, m - K + 1):
            cost = indel_cost(i, j) + W + indel_cost(n - (i + K), m - (j + K))
            best = min(best, cost)
    return best

print(min_cost_transform("abcde", "vwxyz", 3, 2))     # 6
print(min_cost_transform("hello", "hallo", 1, 5))     # 2
print(min_cost_transform("competitive", "cognitive", 3, 1))  # 3
```
**Complexity:** O(|A|·|B|) for the LCS table, O(|A|·|B|) to try all wildcard placements.

### Q22. Maximum sum of B non-overlapping palindromic substrings ("Mirror Beacon", your actual question)
```python
def max_beacon_signal(s, B):
    n = len(s)
    is_pal = [[False] * n for _ in range(n)]
    for i in range(n):
        is_pal[i][i] = True
    for i in range(n - 1):
        if s[i] == s[i + 1]:
            is_pal[i][i + 1] = True
    for length in range(3, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            if s[i] == s[j] and is_pal[i + 1][j - 1]:
                is_pal[i][j] = True

    NEG = float('-inf')
    dp = [[NEG] * (B + 1) for _ in range(n + 1)]
    dp[0][0] = 0
    for i in range(1, n + 1):
        for k in range(B + 1):
            dp[i][k] = dp[i - 1][k]
        for j in range(i):
            if is_pal[j][i - 1]:
                length = i - j
                for k in range(1, B + 1):
                    if dp[j][k - 1] != NEG:
                        dp[i][k] = max(dp[i][k], dp[j][k - 1] + length)
    return max(dp[n][B], 0)
```
**Complexity:** O(N^2) for palindrome table + O(N^2·B) for DP.

### Q23. Standard Edit Distance (insert / delete / replace all cost 1)
```python
def edit_distance(A, B):
    n, m = len(A), len(B)
    dp = [[0] * (m + 1) for _ in range(n + 1)]
    for i in range(n + 1):
        dp[i][0] = i
    for j in range(m + 1):
        dp[0][j] = j
    for i in range(1, n + 1):
        for j in range(1, m + 1):
            if A[i - 1] == B[j - 1]:
                dp[i][j] = dp[i - 1][j - 1]
            else:
                dp[i][j] = 1 + min(dp[i - 1][j], dp[i][j - 1], dp[i - 1][j - 1])
    return dp[n][m]
```
**Complexity:** O(N·M).

### Q26. Longest Palindromic Substring
```python
def longest_palindromic_substring(s):
    n = len(s)
    if n == 0:
        return ""
    start, max_len = 0, 1

    def expand(l, r):
        while l >= 0 and r < n and s[l] == s[r]:
            l -= 1
            r += 1
        return l + 1, r - 1

    for i in range(n):
        l1, r1 = expand(i, i)
        if r1 - l1 + 1 > max_len:
            start, max_len = l1, r1 - l1 + 1
        l2, r2 = expand(i, i + 1)
        if r2 - l2 + 1 > max_len:
            start, max_len = l2, r2 - l2 + 1
    return s[start:start + max_len]
```
**Complexity:** O(N^2) time, O(1) extra space.

```python
def min_cut_palindrome_partition(s):

    is_pal = [[False] * n for _ in range(n)]

        is_pal[i][i] = True
```

### Q44. Egg Drop Problem

```python
def egg_drop(eggs, floors):
    dp = [[0] * (floors + 1) for _ in range(eggs + 1)]
    for j in range(1, floors + 1):
        dp[1][j] = j
    for i in range(1, eggs + 1):
        dp[i][0] = 0
        dp[i][1] = 1
    for i in range(2, eggs + 1):
        for j in range(2, floors + 1):
            dp[i][j] = float('inf')
            for x in range(1, j + 1):
```

### Q45. Matrix Chain Multiplication (min scalar multiplications)

```python
def matrix_chain_order(dims):
    n = len(dims) - 1
    dp = [[0] * n for _ in range(n)]
    for length in range(2, n + 1):
        for i in range(n - length + 1):
            j = i + length - 1
            dp[i][j] = float('inf')
            for k in range(i, j):
                cost = dp[i][k] + dp[k + 1][j] + dims[i] * dims[k + 1] * dims[j + 1]

    return dp[0][n - 1]
```
**Complexity:** O(N^3).

```python
def three_sum(nums):
    nums.sort()
    res = []
    n = len(nums)
    for i in range(n - 2):
        if i > 0 and nums[i] == nums[i - 1]:
            continue
        l, r = i + 1, n - 1
        while l < r:
            s = nums[i] + nums[l] + nums[r]
            if s == 0:
                res.append([nums[i], nums[l], nums[r]])

                    l += 1
                while l < r and nums[r] == nums[r - 1]:
                    r -= 1
                l += 1
                r -= 1

            else:
                r -= 1
```

```python
from collections import deque

def bfs(graph, start):
    visited = {start}
    order = []
    q = deque([start])
    while q:
        node = q.popleft()
        order.append(node)
        for nei in graph[node]:
            if nei not in visited:
                visited.add(nei)
                q.append(nei)
    return order
```

```python
    def find(self, x):
        while self.parent[x] != x:
            self.parent[x] = self.parent[self.parent[x]]
            x = self.parent[x]
        return x

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return
        if self.rank[ra] < self.rank[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        if self.rank[ra] == self.rank[rb]:
            self.rank[ra] += 1
        self.count -= 1

def count_components(n, edges):
    uf = UnionFind(n)
    for a, b in edges:
        uf.union(a, b)
    return uf.count
```
**Complexity:** O(E · α(N)) ≈ O(E).

### Q84. Dijkstra's Shortest Path
```python
import heapq

def dijkstra(graph, n, src):
    dist = [float('inf')] * n
    dist[src] = 0
    heap = [(0, src)]
    while heap:
        d, u = heapq.heappop(heap)
        if d > dist[u]:
            continue
        for v, w in graph[u]:
            nd = d + w
            if nd < dist[v]:
                dist[v] = nd
                heapq.heappush(heap, (nd, v))
    return dist
```

### Q87. Kahn's Topological Sort (return the order)
```python
from collections import deque, defaultdict

def topo_sort(n, edges):
    graph = defaultdict(list)
    indegree = [0] * n
    for u, v in edges:
        graph[u].append(v)
        indegree[v] += 1
    q = deque([i for i in range(n) if indegree[i] == 0])
    order = []
    while q:
        u = q.popleft()
        order.append(u)
        for v in graph[u]:
            indegree[v] -= 1
            if indegree[v] == 0:
                q.append(v)
    return order if len(order) == n else []
```
**Complexity:** O(V + E).

### Q95. Path Sum II (all root-to-leaf paths summing to target)

```python
def path_sum(root, target):
    res = []

    def dfs(node, remaining, path):
        if not node:
            return
        path.append(node.val)
        remaining -= node.val
        if not node.left and not node.right and remaining == 0:
            res.append(list(path))
        else:
            dfs(node.left, remaining, path)
```

### Q98. Min Stack (push/pop/top/getMin all O(1))
```python
class MinStack:
    def __init__(self):
        self.stack = []
        self.min_stack = []

    def push(self, val):
        self.stack.append(val)
        m = val if not self.min_stack else min(val, self.min_stack[-1])
        self.min_stack.append(m)

    def pop(self):
        self.stack.pop()
        self.min_stack.pop()

    def top(self):
        return self.stack[-1]

    def get_min(self):
        return self.min_stack[-1]
```
**Complexity:** O(1) per operation.

### Q99. Merge K Sorted Lists (using heap)

```python
        if node:
            heapq.heappush(heap, (node.val, i, node))
```
