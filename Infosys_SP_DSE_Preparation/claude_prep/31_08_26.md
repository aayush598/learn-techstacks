## 151. Network Delay Time Medium

Given `n` nodes labeled `1` to `n`, a list `times` of directed edges `[u, v, w]` meaning a signal travels from `u` to `v` in `w` time, and a starting node `k`, return the minimum time for a signal starting at `k` to reach all nodes. ```python
import heapq

def networkDelayTime(times, n, k):
    """Minimum time for a signal from node k to reach every node, or -1 if impossible."""
    graph = {i: [] for i in range(1, n + 1)}
    for u, v, w in times:
        graph[u].append((v, w))

    distances = {i: float("inf") for i in range(1, n + 1)}
    distances[k] = 0
    heap = [(0, k)]
    while heap:
        current_distance, node = heapq.heappop(heap)
        if current_distance > distances[node]:
            continue
        for neighbor, weight in graph[node]:
            new_distance = current_distance + weight
            if new_distance < distances[neighbor]:
                distances[neighbor] = new_distance
                heapq.heappush(heap, (new_distance, neighbor))

    max_distance = max(distances.values())
    return max_distance if max_distance != float("inf") else -1

    diagonals_used = set()
    anti_diagonals_used = set()

            return

        return []

    is_prime[0] = False

            for multiple in range(candidate * candidate, n + 1, candidate):
